在各仓实体 artifacts 下新建本轮唯一日志目录，先核输出根，不沿用已有目录；以下 TEST_OUT 为该仓目录，发布脚本内部使用同样唯一日志机制。保留完整输出，命令及 tee 任一失败即停止，不只截取 tail。短测原始输出不进 git，关键命令、判定和退出状态写提交正文。

# 每仓分别在其根目录初始化；仅在代码实施获准后执行
set -euo pipefail
export UV_CACHE_DIR=/home/hongzefu/.cache/uv PYTHONUNBUFFERED=1
ls -ld artifacts
test -d artifacts && test ! -L artifacts
mkdir -p artifacts/checks
TEST_OUT=$(mktemp -d "$PWD/artifacts/checks/success-conditions-v12.XXXXXX")
run_gate() {
  local gate_name="$1" gate_rc tee_rc final_rc
  shift
  set +e
  timeout --signal=TERM --kill-after=5s 275s "$@" 2>&1 | tee "$TEST_OUT/$gate_name.log"
  gate_rc=${PIPESTATUS[0]} tee_rc=${PIPESTATUS[1]}
  set -e
  final_rc=$gate_rc
  if test "$final_rc" -eq 0 && test "$tee_rc" -ne 0; then
    final_rc=$tee_rc
  fi
  printf 'GATE_EXIT name=%s exit=%s tee_exit=%s\nEXIT_CODE=%s\n' \
    "$gate_name" "$gate_rc" "$tee_rc" "$final_rc" | tee -a "$TEST_OUT/$gate_name.log"
  test "$final_rc" -eq 0
}

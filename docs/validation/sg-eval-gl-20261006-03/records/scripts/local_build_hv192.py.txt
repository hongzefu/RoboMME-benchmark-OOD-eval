"""1006 计划第二部分四节 runbook 第 8 步：MemER hard-verify 对拍清单（原侧／新侧同一批 16 任务 × 12 局 = 192）。
源：1004 轮第二档 oracle/gate2 清单（官方 hard 每任务 12 局）；用当前代码 hard-verify builder 逐局反查 (tier, seed, source_episode)；
切 8 片（按行序交替分配，每片 24 局），原侧与新侧共用分片。"""
import json, hashlib, sys
from pathlib import Path
SRC = Path('/nfs/turbo/coe-chaijy-unreplicated/hongzefu/sgeval-20261004/inputs/oracle/gate2/manifest.json')
OUT = Path(sys.argv[1]); OUT.mkdir(parents=True, exist_ok=True)
rows = json.loads(SRC.read_text())['rows']
assert len(rows) == 192 and len({r['key'] for r in rows}) == 192
from robomme_hard.env_record_wrapper import BenchmarkEnvBuilder
bad = 0
for task in sorted({r['task'] for r in rows}):
    b = BenchmarkEnvBuilder(env_id=task, dataset='hard-verify', action_space='joint_angle', max_steps=1300)
    for r in (x for x in rows if x['task'] == task):
        i = b.resolve_identity(r['builder_episode'])
        if (i.get('tier'), int(i['seed']), i.get('source_episode')) != (r['tier'], int(r['seed']), r['source_episode']):
            bad += 1; print('MISMATCH', r['key'], i)
order = sorted(rows, key=lambda r: (r['task'], r['builder_episode']))
shards = {f'{k:02d}': order[k::8] for k in range(8)}
keep = ('task','tier','seed','candidate','builder_episode','source_episode','spec_sha256','key')
for k, v in shards.items():
    (OUT / f'shard-{k}.json').write_text(json.dumps([{kk: x[kk] for kk in keep} for x in v], ensure_ascii=False, indent=1))
man = {'schema': 'sgeval3-hv192/1', 'dataset': 'hard-verify', 'source': str(SRC), 'source_sha256': hashlib.sha256(SRC.read_bytes()).hexdigest(),
       'rule': '16 任务 × 12 局（官方 hard），原侧与新侧共用分片', 'total': 192, 'shards': {k: len(v) for k, v in shards.items()},
       'rows': [dict({kk: x[kk] for kk in keep}, shard=k) for k, v in shards.items() for x in v]}
(OUT / 'manifest.json').write_text(json.dumps(man, ensure_ascii=False, indent=1))
fp = hashlib.sha256(json.dumps(sorted(r['key'] for r in rows)).encode()).hexdigest()
print(f"HV192_MANIFEST={'PASS' if bad == 0 else 'FAIL'} tasks=16 per_task=12 total=192 shards=8x24 identity_mismatch={bad} keys_sha256={fp}")

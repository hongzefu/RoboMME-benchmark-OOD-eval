#!/bin/bash
# 1006 计划第二部分四节 runbook 第 8 步：MemER hard-verify 对拍原侧 1 局 smoke（官方 eval.py MemER 分支经 official_hard_runner.py
# 驱动，兼容层同指纹），hard-verify VideoUnmask，1300 步，模型 seed 7，共享账本 870／50／50／821。用法：smoke_memer_orig.sh <物理卡>
set -u
REPO=/data/hongzefu/robomme_benchmark_MotionJEPANewTask
S=$REPO/artifacts/sg-evaluation/sg-eval-gl-20261006-03/smoke
N=/nfs/turbo/coe-chaijy-unreplicated/hongzefu
CARD=$1
unset CUDA_VISIBLE_DEVICES
export HF_HOME=$N/hf-cache HF_HUB_OFFLINE=1 TRANSFORMERS_OFFLINE=1
export SGEVAL_CLIENT_PY=$REPO/artifacts/sg-evaluation/venvs/client-env/bin/python
export UV_CACHE_DIR=$REPO/artifacts/cache/uv SEAT_XLA_MEM_FRACTION=0.65
CK=/data/hongzefu/robomme_policy_learning-vqa-test/runs/ckpts
ST=$S/stage-memer-orig; [ ! -e $ST ] || { echo "SMOKE_BLOCKED stage_exists $ST"; exit 4; }
echo "SMOKE_START side=orig model=memer card=$CARD head=$(git -C $REPO rev-parse HEAD) $(date -Is)"
cd $REPO && bash scripts/eval-official/run_official_hard.sh --run-name p3-smoke-memer-orig --seat $(printf '%02d' $((90+CARD))) --repo $REPO \
  --stage $ST --shard $S/shard-hv-1.json --policy groundsg --dataset hard-verify --max-steps 1300 \
  --groundsg-variant ground-sg-memer --memer-adapter $REPO/artifacts/sg-eval/ckpt/memer/grounded_subgoal/checkpoint-1300 \
  --groundsg-ckpt $CK/mme_vla_suite/symbolic-grounded-subgoal/79999 \
  --openpi-data-home $REPO/artifacts/v8-evaluation/openpi-data --tokenizer-sha256 8986bb4f423f07f8c7f70d0dbe3526fb2316056c17bae71b1ea975e77a168fc6 \
  --policy-seed 7 --budget-ledger $N/sgeval-20261006-03/budget-ledger.jsonl --trajectory-cap 870 --shared-infra-cap 50 --expired-cap 50 --planned-first-tries 821 \
  --infra-retry-budget 1 --episode-wall 3600 --gpu $CARD --media-root $ST/media
echo "SMOKE_RUN_END side=orig rc=$? $(date -Is)"
find $ST -path '*official*' -name '*.mp4' | head -3
find $ST -name results.jsonl | head -2 | while read r; do tail -1 $r | cut -c1-400; done
uv run --no-sync python scripts/eval-official/lang_io_check.py --require-server-text --root $ST/media 2>&1 | tail -2
uv run --no-sync python scripts/eval-official/trace_arrays_check.py $ST/media 2>&1 | tail -1
find $ST/media -name 'ep*_MemER_log.jsonl' | head -2
echo "SMOKE_DONE side=orig $(date -Is)"

#!/bin/bash
# 主会话事件汇总（只读）：logs/*.log 与 logs/tasks/*.log 每份日志各起一个独立 tail -F（行缓冲过滤、带来源前缀），
# 每 30 秒发现新日志即接上。不用一条 tail 挂多文件（避免 ==> 头部行噪声）。退出时收掉自己起的子进程。
R=/nfs/turbo/coe-chaijy-unreplicated/hongzefu/sgeval-20261006-02
PAT='WORKER_(STOP|END)|TASK_END .*rc=[1-9]|RUN_BLOCKED|INFRA_EXHAUSTED|NO_PROGRESS|OFFICIAL_RENDER=FAIL|OFFICIAL_MEDIA=FAIL|OBSERVER_COMPLETE=FAIL|BUDGET_(WARN|ENFORCEMENT=FAIL)|SEAT_REC_SYNC=FAIL|CHAIN_STOP|EXIT_CODE=[1-9]|Traceback|CUDA error|out of memory|svulkan2|EXCLUSIVE|reset 拒绝|GLIBC|SyntaxError'
declare -A seen
first=1
trap 'kill $(jobs -p) 2>/dev/null' EXIT
while true; do
  for f in $R/logs/*.log $R/logs/tasks/*.log; do
    [ -e "$f" ] || continue
    [ -n "${seen[$f]:-}" ] && continue
    seen[$f]=1
    tag=$(basename "$f" .log)
    ( tail -n $([ $first = 1 ] && echo 0 || echo +1) -F "$f" 2>/dev/null | stdbuf -oL tr '\r' '\n' | grep --line-buffered -aE "$PAT" | stdbuf -oL sed "s|^|[$tag] |" | grep --line-buffered -av "^\[211p00-b4-gate2-pp-s30\] OFFICIAL_RENDER=FAIL" ) &
  done
  first=0; sleep 30
done

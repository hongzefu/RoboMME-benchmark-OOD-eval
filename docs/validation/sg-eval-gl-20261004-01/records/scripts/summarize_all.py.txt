def load(p):
    rows = {}
    for line in open(p):
        if line.strip():
            r = json.loads(line)
            rows[r["key"]] = r
    return rows


def status(r):
    return r.get("result") or r.get("status")


def main():
    print("# 各档成功率总表（收尾时点，自动生成）\n")
    print("| 档 | 站点 | 模型／侧 | 计划局数 | 已有终态 | 成功 | 失败 | 超时 | error | 成功率（按已有行） |")
    print("|---|---|---|---|---|---|---|---|---|---|")
    per_task = {}
    for gate, site, name, path, plan in SETS:
        if not os.path.exists(path):
            print(f"| {gate} | {site} | {name} | {plan} | 文件缺失 | | | | | |")
            continue
        rows = load(path)
        c = collections.Counter(status(r) for r in rows.values())
        n = len(rows)
        print(f"| {gate} | {site} | {name} | {plan} | {n} | {c['success']} | {c['fail']} | {c['timeout']} | "
              f"{n - c['success'] - c['fail'] - c['timeout']} | {100 * c['success'] / max(n, 1):.1f}% |")
        t = collections.defaultdict(lambda: [0, 0])
        for k, r in rows.items():
            task = r.get("task") or k.split("_")[0]
            t[task][0] += 1
            t[task][1] += status(r) == "success"
        per_task[(gate, site, name)] = t
    for gate in ("第二档", "第三档"):
        cols = [k for k in per_task if k[0] == gate]
        tasks = sorted({t for k in cols for t in per_task[k]})
        print(f"\n## {gate}按任务（成功／已跑）\n")
        print("| 任务 | " + " | ".join(f"{k[1]} {k[2]}" for k in cols) + " |")
        print("|---|" + "---|" * len(cols))
        for t in tasks:
            cells = []
            for k in cols:
                n, s = per_task[k].get(t, [0, 0])
                cells.append(f"{s}/{n}" if n else "—")
            print(f"| {t} | " + " | ".join(cells) + " |")


if __name__ == "__main__":
    main()

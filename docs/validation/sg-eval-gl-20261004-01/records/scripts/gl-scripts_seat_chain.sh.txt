#!/bin/bash
# GL 一席的串行链（计划第二部分五「席位队列」）：[第一档两遍] → 第二档本席分片（pair_seat.sh）→ 第三档本席分片（run_eval_gl.sh）。
# 前一段因基础设施非 0 退出（服务起不来、额度耗尽、节点故障：rc 3/4/5/信号）才 CHAIN_STOP；判定结论不参与衔接条件。
# 用法：seat_chain.sh --seat NN --model {oracle|qwenvl|pp} [--gate2 <hard0 分片>] [--gate3 <v9 分片>] [--gate1]
set -u
N=/nfs/turbo/coe-chaijy-unreplicated/hongzefu
C=$N/robomme_benchmark-sgeval
R=$N/sgeval-20261004
FROZEN=$(cat $R/inputs/frozen-commit.txt)
SEAT=""; MODEL=""; G2=""; G3=""; G1=0
while [ $# -gt 0 ]; do case $1 in
  --seat) SEAT=$2; shift 2;; --model) MODEL=$2; shift 2;; --gate2) G2=$2; shift 2;; --gate3) G3=$2; shift 2;; --gate1) G1=1; shift;;
  *) echo "未知参数 $1"; exit 2;; esac; done
echo "SEAT_CHAIN_START seat=$SEAT model=$MODEL gate1=$G1 gate2=${G2:-none} gate3=${G3:-none} host=$(hostname) $(date -Is)"
head=$(git -C $C rev-parse HEAD); dirty=$(git -C $C status --short --ignore-submodules=dirty | wc -l)
[ "$head" = "$FROZEN" ] && [ "$dirty" = 0 ] || { echo "RUN_BLOCKED reason=exec_copy head=$head frozen=$FROZEN dirty=$dirty"; echo "CHAIN_STOP seat=$SEAT reason=exec_copy"; exit 3; }
nvidia-smi --query-gpu=name,driver_version,memory.total --format=csv,noheader | sed 's/^/GPU_INFO /'
export HF_HOME=$N/hf-cache HF_HUB_OFFLINE=1 TRANSFORMERS_OFFLINE=1
export SGEVAL_CLIENT_PY=$N/sg-eval/venvs/client-env/bin/python
export PYTHONPATH=$C/src:$C/third_party/mme-vla/packages/openpi-client/src
export UV_CACHE_DIR=/tmp/uvcache-$SEAT
MMESG="--mmesg-ckpt $N/sg-eval/ckpt/mme/symbolic-grounded-subgoal/79999 --openpi-data-home $N/sg-eval/openpi-data --tokenizer-sha256 8986bb4f423f07f8c7f70d0dbe3526fb2316056c17bae71b1ea975e77a168fc6"
case $MODEL in
  oracle) MODEL_ARGS=(--policy mmesg --mme-variant ground-sg-oracle $MMESG); POL3=(--policies mmesg --mme-variant ground-sg-oracle $MMESG);;
  qwenvl) export SEAT_XLA_MEM_FRACTION=0.65
          QA="--qwenvl-groundsg-adapter $N/sg-eval/ckpt/qwenvl-groundsg/checkpoint-1200"
          MODEL_ARGS=(--policy mmesg --mme-variant ground-sg-qwenvl $QA $MMESG); POL3=(--policies mmesg --mme-variant ground-sg-qwenvl $QA $MMESG);;
  pp) MODEL_ARGS=(--policy pp --pp-ckpt $N/sg-eval/ckpt/pp/ponderpounce-9b-robomme); POL3=(--policies pp --pp-ckpt $N/sg-eval/ckpt/pp/ponderpounce-9b-robomme);;
  *) echo "RUN_BLOCKED reason=model"; exit 2;;
esac
# reset 额度：账本每局记 build 与 reset 各一次（计划预算口径），故按 2 × 局数 + 20
infra_rc() { case $1 in 3|4|5|129|130|143) return 0;; *) return 1;; esac; }
if [ $G1 = 1 ]; then
  for p in v9 x0; do
    bash $R/scripts/gate1_pass.sh new $p; rc=$?
    echo "CHAIN_STEP seat=$SEAT step=gate1-$p rc=$rc $(date -Is)"
  done
fi
if [ -n "$G2" ]; then
  bash $C/scripts/eval-official/pair_seat.sh --run-name g2-$MODEL-s$SEAT --seat $SEAT --repo $C --stage $R/gate2-$MODEL --shard $G2 \
    "${MODEL_ARGS[@]}" --reset-budget $(( 2 * $(grep -c '"key"' $G2) + 20 )) --infra-retry-budget ${G2_INFRA:-3} --orig-infra-retry-budget ${G2_INFRA:-3}
  rc=$?; echo "CHAIN_STEP seat=$SEAT step=gate2 rc=$rc $(date -Is)"
  if infra_rc $rc; then echo "CHAIN_STOP seat=$SEAT step=gate2 rc=$rc"; exit $rc; fi
fi
if [ -n "$G3" ]; then
  bash $C/scripts/eval-official/run_eval_gl.sh --run-name g3-$MODEL-s$SEAT --seat $SEAT --repo $C --stage $R/gate3-$MODEL --shard $G3 \
    --dataset test-hard --max-steps 1600 --strict-cap "${POL3[@]}" --media-root $R/media \
    --reset-budget $(( 2 * $(grep -c '"key"' $G3) + 20 )) --infra-retry-budget ${G3_INFRA:-3}
  rc=$?; echo "CHAIN_STEP seat=$SEAT step=gate3 rc=$rc $(date -Is)"
  if infra_rc $rc; then echo "CHAIN_STOP seat=$SEAT step=gate3 rc=$rc"; exit $rc; fi
fi
echo "SEAT_CHAIN_DONE seat=$SEAT model=$MODEL $(date -Is)"

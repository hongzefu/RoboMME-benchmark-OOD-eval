#!/bin/bash
# GL 登录节点 tmux 内：launch_seat.sh <JobID> <日志名> <seat_chain.sh 参数…>
# 等占位作业 RUNNING 后清 SLURM_* 变量，srun 进该占位作业跑 seat_chain.sh（只有这一步带 --gpu_cmode=shared）。
JOB=$1; NAME=$2; shift 2
R=/nfs/turbo/coe-chaijy-unreplicated/hongzefu/sgeval-20261004
mkdir -p $R/logs
while [ "$(squeue -h -j $JOB -o %T 2>/dev/null)" != "RUNNING" ]; do
  st=$(squeue -h -j $JOB -o %T 2>/dev/null); [ -z "$st" ] && { echo "错误: 作业 $JOB 不在队列"; echo "EXIT_CODE=3"; exit 3; }
  sleep 60
done
echo "SEAT_LAUNCH job=$JOB name=$NAME node=$(squeue -h -j $JOB -o %N) $(date -Is)"
for v in $(env | grep -o '^SLURM_[A-Z_]*'); do unset "$v"; done
cd /tmp
srun --jobid=$JOB --overlap --exact --ntasks=1 --cpus-per-task=4 --gpu_cmode=shared bash $R/scripts/seat_chain.sh "$@" 2>&1 | tee -a $R/logs/$NAME.log
echo "EXIT_CODE=${PIPESTATUS[0]}" | tee -a $R/logs/$NAME.log
